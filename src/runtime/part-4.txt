function enterNode(type){

}

function enemyProfile(name){
 if(name.includes('哥布林')) return {kind:'goblin',trait:'偷窃',armor:0,escape:0};
 if(name.includes('魔像')||name.includes('石像')) return {kind:'golem',trait:'护甲',armor:3,escape:0};
 if(name.includes('蝙蝠')) return {kind:'bat',trait:'迅捷',armor:0,escape:0};
 if(name.includes('蘑菇')) return {kind:'mushroom',trait:'孢子',armor:0,escape:0};
 if(name.includes('宝箱')) return {kind:'mimic',trait:'逃跑',armor:0,escape:4};
 if(name.includes('酋长')) return {kind:'boss_goblin',trait:'掠夺',armor:1,escape:0};
 if(name.includes('水晶魔像')) return {kind:'boss_golem',trait:'坚甲',armor:5,escape:0};
 if(name.includes('巨型史莱姆')) return {kind:'boss_slime',trait:'分裂',armor:0,escape:0};
 return {kind:'slime',trait:'普通',armor:0,escape:0};
}
function applyEnemyStartMechanic(){
 S.enemyTurnCount=0;S.enemyRage=false;
 if(S.enemyKind==='bat'){
  S.enemyDamage=Math.max(1,S.enemyDamage-2);
  ui.tip.textContent='洞穴蝙蝠会连续攻击两次，但每次伤害较低。';
 }
 if(S.enemyKind==='golem'||S.enemyKind==='boss_golem'){
  ui.tip.textContent=`${S.enemy} 拥有 ${S.enemyArmor} 点护甲，普通落槽伤害会被减免。`;
 }
 if(S.enemyKind==='mimic'){
  ui.tip.textContent=`宝箱怪将在 ${S.enemyEscapeTurn} 回合后逃跑，击败可获得额外金币。`;
 }
}
function enemyMechanicBeforeAttack(){
 if(S.enemyKind==='goblin'||S.enemyKind==='boss_goblin'){
  const steal=Math.min(S.coins,S.enemyKind==='boss_goblin'?8:4);
  if(steal>0){
   S.coins-=steal;
   ui.tip.innerHTML=`${S.enemy} 偷走了 <b style="color:#ffd166">${steal}</b> 金币。`;
  }
 }
 if(S.enemyKind==='mushroom'){
  const candidates=S.pegs.filter(p=>!p.frozen).sort(()=>Math.random()-.5).slice(0,3);
  candidates.forEach(p=>p.frozen=true);
  if(candidates.length) ui.tip.textContent='蘑菇孢子冻结了 3 个钉子，本回合命中它们不会得分。';
 }
 if((S.enemyKind==='boss_slime')&&S.enemyTurnCount===2){
  S.ehp=Math.min(S.emax,S.ehp+Math.floor(S.emax*.18));
  ui.tip.textContent='巨型史莱姆分裂并恢复了生命。';
 }
}
function enemyMechanicAfterAttack(){
 if(S.enemyKind==='mushroom') S.pegs.forEach(p=>p.frozen=false);
 if((S.enemyKind==='golem'||S.enemyKind==='boss_golem') && S.ehp<S.emax/2 && !S.enemyRage){
  S.enemyRage=true;S.enemyDamage+=3;
  ui.tip.textContent='魔像外壳破裂，攻击力上升！';
 }
}

function startBattle(type='battle',advance=true){
 if(advance)S.stage++;S.node++;S.battleType=type;const boss=type==='boss'||S.stage%5===0,elite=type==='elite';
 const names=boss?['巨型史莱姆','哥布林酋长','水晶魔像']:elite?['精英骷髅','狂暴食人花','重甲魔像']:['持盾哥布林','洞穴蝙蝠','蘑菇怪','骷髅兵'];
 S.enemy=rand(names);
 const profile=enemyProfile(S.enemy);
 S.enemyKind=profile.kind;S.enemyTrait=profile.trait;S.enemyArmor=profile.armor;S.enemyEscapeTurn=profile.escape;
 const factor=boss?1.6:elite?1.28:1;S.emax=Math.floor((48+S.stage*27+S.stage*S.stage*2.2)*factor);S.ehp=S.emax;
 S.enemyDamage=Math.floor((6+S.stage*2.05)*(boss?1.35:elite?1.18:1));S.firstShot=true;S.turnInBattle=0;S.gameOver=false;
 S.magazine=shuffledCopy(S.deck);S.reloadCount=0;
 ui.overlay.classList.remove('show');buildBoard();updateUI();ui.tip.textContent=boss?'Boss 战！弹夹已随机装填。':elite?'精英战开始，弹夹已随机装填。':'新的敌人出现，弹夹已随机装填。左右边界支持跨屏。';save();
}
