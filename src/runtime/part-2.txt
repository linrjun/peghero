function shuffledCopy(arr){
 const out=[...arr];
 for(let i=out.length-1;i>0;i--){
  const j=Math.floor(Math.random()*(i+1));
  [out[i],out[j]]=[out[j],out[i]];
 }
 return out;
}

async function enableMotion(){
 try{
  if(typeof DeviceMotionEvent==='undefined'){
   ui.motionBtn.textContent='设备不支持摇晃';
   ui.tip.textContent='当前浏览器没有提供运动传感器。';
   return;
  }
  if(typeof DeviceMotionEvent.requestPermission==='function'){
   const result=await DeviceMotionEvent.requestPermission();
   if(result!=='granted')throw new Error('permission denied');
  }
  if(!S.motionListening){
   window.addEventListener('devicemotion',onDeviceMotion,{passive:true});
   S.motionListening=true;
  }
  S.motionEnabled=!S.motionEnabled;
  ui.motionBtn.textContent=`摇晃辅助：${S.motionEnabled?'开':'关'}`;
 ui.pegBtn.textContent=`钉子袋 ${S.pegBag.length}/${S.pegBagCapacity}`;
 ui.pegBtn.classList.toggle('active',!!S.selectedPegItem);
 updateBuildUI();updateMagazineUI();
  ui.tip.textContent=S.motionEnabled
   ?'摇晃辅助已开启：仅在弹珠低速卡住时生效，每颗弹珠最多一次。'
   :'摇晃辅助已关闭。';
 }catch(e){
  ui.motionBtn.textContent='摇晃权限未开启';
  ui.tip.textContent='浏览器未授予运动传感器权限，请点击按钮后允许访问。';
 }
}
function onDeviceMotion(e){
 if(!S.motionEnabled||!S.ball||!S.ball.active||S.nudgeUsed||S.stuckTime<.9)return;
 const a=e.accelerationIncludingGravity||e.acceleration;
 if(!a)return;
 const now={x:Number(a.x)||0,y:Number(a.y)||0,z:Number(a.z)||0};
 if(!S.lastAccel){S.lastAccel=now;return}
 const dx=now.x-S.lastAccel.x,dy=now.y-S.lastAccel.y,dz=now.z-S.lastAccel.z;
 S.lastAccel=now;
 const force=Math.hypot(dx,dy,dz);
 if(force<10.5||performance.now()-S.lastNudgeAt<1200)return;
 rescueNudge(dx,dy,force);
}
