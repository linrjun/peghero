function applySpecialPegEffect(p,b){
 if(p.type==='normal')return;
 if(p.type==='spring'){
  const speed=Math.max(240,Math.hypot(b.vx,b.vy));
  const nx=(b.x-p.x)/(Math.hypot(b.x-p.x,b.y-p.y)||1);
  const ny=(b.y-p.y)/(Math.hypot(b.x-p.x,b.y-p.y)||1);
  const boost=buildTier('combo')===2?1.48:buildTier('combo')===1?1.34:1.22;
  b.vx=nx*speed*boost;b.vy=ny*speed*boost;
  showFloatingText(p.x,p.y-12,'弹力！','#7ee787');
 }
 if(p.consumed)return;
 p.consumed=true;
 if(p.type==='bombPeg'){
  dealInstantDamage(10+(buildTier('bomb')===2?13:buildTier('bomb')===1?5:0),p.x,p.y,'炸弹钉');p.type='normal';
 }
 if(p.type==='goldPeg'){
  const gain=buildTier('wealth')===2?10:buildTier('wealth')===1?6:4;
  S.coins+=gain;showFloatingText(p.x,p.y-12,`+${gain} 金币`,'#ffd166');updateUI();
 }
 if(p.type==='healPeg'){
  const before=S.php;S.php=Math.min(S.pmax,S.php+5+buildTier('life'));
  showFloatingText(p.x,p.y-12,`+${Math.ceil(S.php-before)} 生命`,'#72e6a8');updateUI();
 }
 if(p.type==='amplifier'){
  S.shotDamage+=8;showFloatingText(p.x,p.y-12,'本球 +8','#c59aff');
 }
 if(p.type==='flamePeg'){
  const dmg=buildTier('fire')===2?8:4;dealInstantDamage(dmg,p.x,p.y,'燃烧钉');
 }
 if(p.type==='conductor'){
  S.hitCount+=2;S.shotDamage+=2;showFloatingText(p.x,p.y-12,'连击 +2','#80e9ff');
 }
 if(p.type==='bloodPeg'){
  const heal=3+buildTier('life'),before=S.php;S.php=Math.min(S.pmax,S.php+heal);
  dealInstantDamage(3+(buildTier('life')===2?2:0),p.x,p.y,'血泉');
  if(S.php>before)showFloatingText(p.x+18,p.y,`+${Math.ceil(S.php-before)} 生命`,'#72e6a8');
 }
}
