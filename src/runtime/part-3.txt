 if(dot<0){b.vx-=1.82*dot*nx;b.vy-=1.82*dot*ny;b.vx*=.995;b.vy*=.995}

function update(dt){
 S.particles.forEach(p=>{p.x+=p.vx*dt;p.y+=p.vy*dt;if(!p.isText)p.vy+=70*dt;p.life-=dt});
 S.particles=S.particles.filter(p=>p.life>0);S.pegs.forEach(p=>p.pulse=Math.max(0,p.pulse-dt*4));
 const b=S.ball;if(!b||!b.active)return;
 const speed=Math.hypot(b.vx,b.vy);
 const slotTop=H-Y(142);
 if(speed<46&&b.y<slotTop-12){
  S.stuckTime+=dt;
  if(S.motionEnabled&&!S.nudgeUsed&&S.stuckTime>.9&&S.stuckTime-dt<=.9){
   ui.tip.textContent='弹珠似乎卡住了，现在摇晃手机可脱困一次。';
  }
 }else S.stuckTime=Math.max(0,S.stuckTime-dt*2.5);
 b.trail.push({x:b.x,y:b.y,life:.35});if(b.trail.length>24)b.trail.shift();
 const steps=3,sub=dt/steps;
 for(let i=0;i<steps;i++){
  if(hasRelic('magnet')){
   let nearest=null,nd=95;
   for(const p of S.pegs){if(b.already.has(p.id))continue;const d=Math.hypot(p.x-b.x,p.y-b.y);if(d<nd){nd=d;nearest=p}}
   if(nearest){b.vx+=(nearest.x-b.x)*.45*sub;b.vy+=(nearest.y-b.y)*.45*sub}
  }
  b.vy+=S.gravity*sub;b.x+=b.vx*sub;b.y+=b.vy*sub;
  // 左右跨屏，不再反弹
  if(b.x+b.r<0){
    b.x=W+b.r;
  }else if(b.x-b.r>W){
    b.x=-b.r;
  }
  S.pegs.forEach(p=>collide(b,p));
 }
 const top=H-Y(142);if(b.y>top&&b.vy>0){const sw=W/5,idx=Math.max(0,Math.min(4,Math.floor(b.x/sw)));finishShot(S.slots[idx]+S.multBonus)}
}
